Network Data

네트워크 전송용 입력 데이터

[Flags]
public enum HeroInputButtons
{
    None = 0,
    Attack = 1 << 0,
    AttackHold = 1 << 1,
    Guard = 1 << 2,
    Dash = 1 << 3,
    FocusMode = 1 << 4,
    FocusChange = 1 << 5,
    Ultimate = 1 << 6,
    AttackGuard = 1 << 7
}

public struct HeroInputData : INetworkInput
{
    public float Horizontal;
    public float Vertical;
    public HeroInputButtons Buttons;
    public float ClientTimeStamp;
    public float SimulationTimeStamp;

    public Vector2 Move => new(Horizontal, Vertical);

    public bool Has(HeroInputButtons button)
        => (Buttons & button) == button;

    public void Set(HeroInputButtons button, bool pressed)
    {
        if (pressed) Buttons |= button;
        else Buttons &= ~button;
    }
}

포인트

  • 버튼 상태를 enum flag로 압축
  • 이동축과 버튼 입력을 하나의 구조체로 통합
  • 타임스탬프를 포함해 네트워크 지연 분석 가능

실무 의도

액션 게임 입력은 매 tick 전송될 수 있으므로 데이터 구조가 단순해야 합니다. 버튼 플래그를 사용하면 전송량과 분기 복잡도를 줄일 수 있습니다.

입력 장치 추상화

public interface IHeroInputSource
{
    float Horizontal { get; }
    float Vertical { get; }
    bool AttackPressed { get; }
    bool AttackHeld { get; }
    bool GuardPressed { get; }
    bool DashPressed { get; }
    bool FocusModePressed { get; }
    bool FocusChangePressed { get; }
    bool UltimatePressed { get; }
}

public sealed class KeyboardHeroInputSource : MonoBehaviour, IHeroInputSource
{
    [SerializeField] private InputActionReference move;
    [SerializeField] private InputActionReference attack;
    [SerializeField] private InputActionReference guard;
    [SerializeField] private InputActionReference dash;
    [SerializeField] private InputActionReference ultimate;

    public float Horizontal => move.action.ReadValue<Vector2>().x;
    public float Vertical => move.action.ReadValue<Vector2>().y;
    public bool AttackPressed => attack.action.WasPressedThisFrame();
    public bool AttackHeld => attack.action.IsPressed();
    public bool GuardPressed => guard.action.IsPressed();
    public bool DashPressed => dash.action.WasPressedThisFrame();
    public bool FocusModePressed => false;
    public bool FocusChangePressed => false;
    public bool UltimatePressed => ultimate.action.WasPressedThisFrame();
}

리뷰 포인트

입력 장치를 인터페이스로 분리하면 키보드, 패드, 모바일 터치 입력을 같은 수집 흐름으로 처리할 수 있습니다. 네트워크 시스템은 실제 입력 장치 구현을 알 필요가 없습니다.

Input Collector

로컬 입력 수집과 동시 입력 규칙

public sealed class HeroInputCollector
{
    private readonly IReadOnlyList<IHeroInputSource> _sources;

    public HeroInputCollector(IReadOnlyList<IHeroInputSource> sources)
    {
        _sources = sources;
    }

    public HeroInputData Collect(float simulationTime)
    {
        var data = new HeroInputData
        {
            ClientTimeStamp = Time.time,
            SimulationTimeStamp = simulationTime
        };

        foreach (var source in _sources)
        {
            data.Horizontal = source.Horizontal;
            data.Vertical = source.Vertical;
            data.Set(HeroInputButtons.Attack, source.AttackPressed);
            data.Set(HeroInputButtons.AttackHold, source.AttackHeld);
            data.Set(HeroInputButtons.Guard, source.GuardPressed);
            data.Set(HeroInputButtons.Dash, source.DashPressed);
            data.Set(HeroInputButtons.FocusMode, source.FocusModePressed);
            data.Set(HeroInputButtons.FocusChange, source.FocusChangePressed);
            data.Set(HeroInputButtons.Ultimate, source.UltimatePressed);
            data.Set(HeroInputButtons.AttackGuard,
                source.AttackPressed && source.GuardPressed);
        }

        return data;
    }
}

동시 입력 처리

공격+가드처럼 별도 액션으로 해석해야 하는 조합 입력은 수집 단계에서 명확한 플래그로 변환합니다.

확장성

입력 소스가 여러 개여도 Collector가 동일한 네트워크 입력 데이터로 정규화합니다.

Network Bridge

입력 권한 검증과 네트워크 전송

public sealed class NetworkHeroInput : NetworkBehaviour
{
    private HeroInputCollector _collector;

    public override void Spawned()
    {
        var sources = GetComponents<IHeroInputSource>();
        _collector = new HeroInputCollector(sources);
    }

    public override void OnInput(NetworkRunner runner, NetworkInput input)
    {
        if (!Object.HasInputAuthority)
            return;

        var data = _collector.Collect(runner.SimulationTime);
        input.Set(data);
    }
}

public readonly struct HeroInputSyncContext
{
    public float ClientTimeStamp { get; }
    public float SimulationTimeStamp { get; }
    public float LocalReceiveTime { get; }

    public HeroInputSyncContext(HeroInputData data, float receiveTime)
    {
        ClientTimeStamp = data.ClientTimeStamp;
        SimulationTimeStamp = data.SimulationTimeStamp;
        LocalReceiveTime = receiveTime;
    }

    public float EstimatedLatency => LocalReceiveTime - ClientTimeStamp;
}

포인트

HasInputAuthority를 확인해 로컬 플레이어만 입력을 생성합니다. 네트워크 수신 측에서는 입력에 포함된 타임스탬프로 지연 추정값을 만들고, 스킬/애니메이션 보정에 전달할 수 있습니다.

Input Processor

수신 입력을 캐릭터 명령으로 변환

public sealed class NetworkHeroController : NetworkBehaviour
{
    [SerializeField] private HeroController heroController;
    [SerializeField] private HeroActionRouter actionRouter;

    public override void FixedUpdateNetwork()
    {
        if (!GetInput(out HeroInputData input))
            return;

        var context = new HeroInputSyncContext(input, Time.time);

        heroController.Move(input.Move);

        if (input.Has(HeroInputButtons.Dash))
            actionRouter.ExecuteDash(context);

        if (input.Has(HeroInputButtons.Guard))
            actionRouter.SetGuard(true, context);
        else
            actionRouter.SetGuard(false, context);

        if (input.Has(HeroInputButtons.AttackGuard))
            actionRouter.ExecuteCounter(context);
        else if (input.Has(HeroInputButtons.Attack))
            actionRouter.ExecuteAttack(context);

        if (input.Has(HeroInputButtons.AttackHold))
            actionRouter.ChargeAttack(context);

        if (input.Has(HeroInputButtons.Ultimate))
            actionRouter.ExecuteUltimate(context);
    }
}

분기 우선순위

공격+가드처럼 더 구체적인 조합 입력을 일반 공격보다 먼저 처리해 의도하지 않은 중복 실행을 막습니다.

책임 분리

NetworkHeroController는 입력을 해석하고, 실제 전투 동작 실행은 HeroActionRouter로 위임합니다.

Hero Action Router

실제 액션 실행 계층

public sealed class HeroActionRouter : MonoBehaviour
{
    [SerializeField] private HeroMovement movement;
    [SerializeField] private HeroGuard guard;
    [SerializeField] private HeroAttack attack;
    [SerializeField] private HeroSkill ultimate;

    public void ExecuteDash(HeroInputSyncContext context)
    {
        if (!movement.CanDash)
            return;

        movement.Dash(context.EstimatedLatency);
    }

    public void SetGuard(bool active, HeroInputSyncContext context)
    {
        if (active) guard.Enter(context.EstimatedLatency);
        else guard.Exit();
    }

    public void ExecuteAttack(HeroInputSyncContext context)
    {
        if (!attack.CanAttack)
            return;

        attack.Execute(new AttackCommand
        {
            Latency = context.EstimatedLatency,
            IsCharged = false
        });
    }

    public void ChargeAttack(HeroInputSyncContext context)
    {
        attack.Charge(context.EstimatedLatency);
    }

    public void ExecuteUltimate(HeroInputSyncContext context)
    {
        if (ultimate.IsReady)
            ultimate.Execute(context.EstimatedLatency);
    }
}

실제 동작까지 보여주는 이유

입력 전송 코드만 있으면 네트워크 API 사용 예제로 끝납니다. 액션 실행 계층까지 보여주면 입력이 실제 캐릭터 행동으로 변환되는 구조와 전투 시스템 연결 방식을 설명할 수 있습니다.

Movement Example

이동/대시 실제 처리 예시

public sealed class HeroMovement : MonoBehaviour
{
    [SerializeField] private CharacterController controller;
    [SerializeField] private float moveSpeed = 6f;
    [SerializeField] private float dashPower = 12f;
    [SerializeField] private float dashCooldown = 0.35f;

    private Vector3 _lastMoveDirection;
    private float _nextDashTime;

    public bool CanDash => Time.time >= _nextDashTime;

    public void Move(Vector2 input)
    {
        var direction = new Vector3(input.x, 0f, input.y);
        if (direction.sqrMagnitude > 1f)
            direction.Normalize();

        if (direction.sqrMagnitude > 0.001f)
            _lastMoveDirection = direction;

        controller.Move(direction * moveSpeed * Time.deltaTime);
    }

    public void Dash(float latency)
    {
        _nextDashTime = Time.time + dashCooldown;

        var compensation = Mathf.Clamp(latency, 0f, 0.12f);
        var distance = dashPower * (Time.deltaTime + compensation);
        controller.Move(_lastMoveDirection * distance);
    }
}

포인트

  • 입력 벡터 정규화
  • 마지막 이동 방향을 대시 방향으로 재사용
  • 지연값을 제한적으로 보정에 사용

주의점

실제 상용 코드에서는 예측/보정 정책, 서버 권한, 충돌 처리 기준에 맞춰 대시 보정 범위를 더 엄격하게 관리해야 합니다.

Attack Example

공격 실행과 명령 객체

public readonly struct AttackCommand
{
    public float Latency { get; init; }
    public bool IsCharged { get; init; }
}

public sealed class HeroAttack : MonoBehaviour
{
    [SerializeField] private Animator animator;
    [SerializeField] private HitBoxController hitBox;
    [SerializeField] private float attackCooldown = 0.25f;

    private float _nextAttackTime;
    private bool _charging;

    public bool CanAttack => Time.time >= _nextAttackTime;

    public void Execute(AttackCommand command)
    {
        _nextAttackTime = Time.time + attackCooldown;
        _charging = false;

        animator.SetTrigger(command.IsCharged ? "ChargedAttack" : "Attack");
        hitBox.Open(new HitBoxOpenContext
        {
            StartOffset = Mathf.Clamp(command.Latency, 0f, 0.1f),
            IsCharged = command.IsCharged
        });
    }

    public void Charge(float latency)
    {
        if (_charging)
            return;

        _charging = true;
        animator.SetBool("Charging", true);
    }
}

면접 설명 포인트

입력 처리 계층이 직접 애니메이터와 히트박스를 제어하지 않고, 공격 실행 계층에 명령을 전달합니다. 이렇게 하면 입력 소스가 네트워크인지 AI인지와 무관하게 공격 시스템을 재사용할 수 있습니다.

테스트 가능성과 면접 어필 포인트

테스트 가능한 항목

  • 버튼 플래그 조합이 올바르게 생성되는지
  • 권한이 없는 클라이언트가 입력을 만들지 않는지
  • 입력 데이터가 컨트롤러 명령으로 올바르게 변환되는지
  • 공격+가드 조합 입력이 일반 공격보다 먼저 처리되는지
  • 지연값이 제한 범위 안에서만 보정에 쓰이는지

면접에서 설명할 내용

  • Input Authority와 State Authority 구분
  • 네트워크 입력과 RPC를 어떤 기준으로 나눌지
  • 액션 게임에서 입력 지연을 측정하는 방식
  • 캐릭터 컨트롤러를 네트워크와 분리하는 이유
  • 입력 처리부터 실제 동작까지의 호출 흐름
