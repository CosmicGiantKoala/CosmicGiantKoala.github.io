public readonly struct SkillExecuteResult
{
    public bool Success { get; init; }
    public string Reason { get; init; }

    public static SkillExecuteResult Ok() => new() { Success = true };
    public static SkillExecuteResult Fail(string reason) => new() { Success = false, Reason = reason };
}

Skill Unit

스킬 실행 단위

public abstract class HeroSkill
{
    public SkillData Data { get; }
    private float _nextAvailableTime;

    protected HeroSkill(SkillData data)
    {
        Data = data;
    }

    public SkillExecuteResult TryExecute(SkillContext context)
    {
        if (Time.time < _nextAvailableTime)
            return SkillExecuteResult.Fail("Cooldown");

        if (!CanPayCost(context.Status))
            return SkillExecuteResult.Fail("NotEnoughResource");

        if (!CanExecuteByState(context.Status))
            return SkillExecuteResult.Fail("InvalidState");

        PayCost(context.Status);
        _nextAvailableTime = Time.time + Data.Cooldown;
        ExecuteCore(context);
        return SkillExecuteResult.Ok();
    }

    protected abstract void ExecuteCore(SkillContext context);
}

실무 의도

쿨타임, 비용, 상태 검증은 공통 흐름으로 처리하고, 실제 스킬 효과만 하위 클래스에서 구현합니다.

State Validation

상태 기반 실행 제한

private static bool CanExecuteByState(IHeroStatus status)
{
    if (status.IsDead) return false;
    if (status.IsStunned) return false;
    if (status.IsDown) return false;
    if (status.IsCasting) return false;
    return true;
}

private static bool CanPayCost(IHeroStatus status, SkillData data)
{
    return data.CostType switch
    {
        SkillCostType.None => true,
        SkillCostType.Stamina => status.Stamina >= data.CostValue,
        SkillCostType.SkillPoint => status.SkillPoint >= data.CostValue,
        _ => false
    };
}

private static void PayCost(IHeroStatus status, SkillData data)
{
    if (data.CostType == SkillCostType.Stamina)
        status.ConsumeStamina(data.CostValue);
    else if (data.CostType == SkillCostType.SkillPoint)
        status.ConsumeSkillPoint(data.CostValue);
}

Concrete Skill

공격 스킬 구현 예시

public sealed class StrikeSkill : HeroSkill
{
    private readonly ISkillEventBus _eventBus;

    public StrikeSkill(SkillData data, ISkillEventBus eventBus) : base(data)
    {
        _eventBus = eventBus;
    }

    protected override void ExecuteCore(SkillContext context)
    {
        _eventBus.Publish(new SkillStartedEvent
        {
            Caster = context.Caster,
            AnimationTrigger = Data.AnimationTrigger,
            EffectId = Data.EffectId
        });

        _eventBus.Publish(new HitBoxRequestedEvent
        {
            Caster = context.Caster,
            ProfileId = Data.HitBoxProfileId,
            AimDirection = context.AimDirection,
            Latency = context.NetworkLatency
        });
    }
}

Ability Loadout

영웅 어빌리티별 스킬 세트

public enum HeroAbilityType { Boxer, Ninja, Wizard }

public sealed class AbilityLoadout
{
    private readonly Dictionary<SkillSlot, HeroSkill> _skills = new();

    public HeroAbilityType Type { get; }

    public AbilityLoadout(HeroAbilityType type, IEnumerable<HeroSkill> skills)
    {
        Type = type;
        foreach (var skill in skills)
            _skills[skill.Data.Slot] = skill;
    }

    public bool TryGetSkill(SkillSlot slot, out HeroSkill skill)
        => _skills.TryGetValue(slot, out skill);
}

public sealed class HeroAbilityManager
{
    private AbilityLoadout _current;

    public void ChangeAbility(AbilityLoadout loadout)
    {
        _current = loadout;
        AbilityChanged?.Invoke(loadout.Type);
    }

    public event Action<HeroAbilityType> AbilityChanged;
}

실무 의도

캐릭터의 어빌리티가 바뀌면 스킬 구성이 통째로 교체됩니다. 네트워크 동기화 대상은 AbilityType만 두고, 각 클라이언트가 같은 Loadout을 구성할 수 있습니다.

Skill Runner

입력에서 스킬 실행까지 연결

public sealed class HeroSkillRunner
{
    private readonly HeroAbilityManager _abilityManager;
    private readonly IHeroStatus _status;

    public SkillExecuteResult TryRun(
        SkillSlot slot,
        HeroInputSyncContext syncContext,
        Vector3 aimDirection)
    {
        if (!_abilityManager.Current.TryGetSkill(slot, out var skill))
            return SkillExecuteResult.Fail("SkillNotFound");

        var context = new SkillContext
        {
            Caster = _heroController,
            Status = _status,
            AimDirection = aimDirection,
            NetworkLatency = syncContext.EstimatedLatency,
            IsLocalPlayer = syncContext.IsLocal
        };

        return skill.TryExecute(context);
    }
}

NetworkInput과 연결

입력 샘플의 HeroActionRouter가 SkillRunner를 호출하면 입력-스킬 흐름이 완성됩니다.
