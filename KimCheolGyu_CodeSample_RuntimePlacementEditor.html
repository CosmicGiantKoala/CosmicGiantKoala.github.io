Raycast Utility

포인터 기반 Raycast와 우선순위 선택

public sealed class PlacementInputRaycaster : MonoBehaviour
{
    [SerializeField] private Camera raycastCamera;
    [SerializeField] private float maxDistance = 50f;
    [SerializeField] private float minDistance = 0.5f;

    private readonly RaycastHit[] _hits = new RaycastHit[10];

    public bool TryGetHit(
        Vector2 screenPosition,
        LayerMask targetMask,
        int priorityLayer,
        out RaycastHit result)
    {
        var ray = raycastCamera.ScreenPointToRay(screenPosition);
        var count = Physics.RaycastNonAlloc(ray, _hits, maxDistance, targetMask);

        if (count == 0)
        {
            result = default;
            return false;
        }

        var validHits = _hits
            .Take(count)
            .Where(hit => hit.collider != null && hit.distance > minDistance)
            .OrderBy(hit => hit.distance)
            .ToArray();

        if (validHits.Length == 0)
        {
            result = default;
            return false;
        }

        var priorityHit = validHits.FirstOrDefault(
            hit => hit.transform.gameObject.layer == priorityLayer);

        result = priorityHit.collider != null ? priorityHit : validHits[0];
        return true;
    }
}

실무 의도

RaycastNonAlloc으로 반복 입력 중 GC를 줄이고, 우선순위 레이어로 겹친 오브젝트 중 의도한 편집 대상을 안정적으로 선택합니다.

UI Guard

UI 입력과 월드 입력 충돌 방지

public sealed class PointerUiGuard
{
    private readonly EventSystem _eventSystem;
    private readonly List<RaycastResult> _results = new();

    public PointerUiGuard(EventSystem eventSystem)
    {
        _eventSystem = eventSystem;
    }

    public bool IsPointerOnUI(Vector2 screenPosition)
    {
        if (_eventSystem == null)
            return false;

        _results.Clear();

        var eventData = new PointerEventData(_eventSystem)
        {
            position = screenPosition
        };

        _eventSystem.RaycastAll(eventData, _results);
        return _results.Count > 0;
    }
}

public sealed class EditorPointerInput
{
    public Vector2 Position => Mouse.current.position.ReadValue();
    public Vector2 Delta => Mouse.current.delta.ReadValue();
    public bool Pressed => Mouse.current.leftButton.wasPressedThisFrame;
    public bool Released => Mouse.current.leftButton.wasReleasedThisFrame;
    public bool Cancelled => Keyboard.current.escapeKey.wasPressedThisFrame;
}

Placement Validation

배치 영역 검증과 자기 자신 무시

public sealed class PlacementAreaResolver
{
    private readonly PlacementInputRaycaster _raycaster;
    private readonly PointerUiGuard _uiGuard;
    private readonly LayerMask _areaMask;

    public bool TryGetPlacementArea(
        Vector2 screenPosition,
        IEditableProp ignoredProp,
        out IPlaceableArea area)
    {
        area = null;

        if (_uiGuard.IsPointerOnUI(screenPosition))
            return false;

        if (!_raycaster.TryGetHit(screenPosition, _areaMask, -1, out var hit))
            return false;

        if (!hit.collider.TryGetComponent(out IPlaceableArea candidate))
            return false;

        if (candidate.IsChildAreaOf(out var parent) && parent == ignoredProp)
            return false;

        area = candidate;
        return true;
    }
}

문제

이동 중인 오브젝트가 자기 자신의 배치 영역을 다시 감지하면 배치 검증이 흔들릴 수 있습니다.

해결

현재 편집 중인 오브젝트를 필터링하고, UI 위 포인터 입력을 먼저 차단해 월드 조작과 UI 조작이 충돌하지 않게 했습니다.

Edit Session

편집 세션과 취소 복원

public sealed class PropEditSession
{
    public IEditableProp Target { get; }
    public PlacementSnapshot Snapshot { get; }
    public bool IsActive { get; private set; }

    public PropEditSession(IEditableProp target)
    {
        Target = target;
        Snapshot = target.Capture();
        IsActive = true;
        target.Select();
    }

    public void Apply()
    {
        if (!IsActive)
            return;

        IsActive = false;
        Target.Deselect();
    }

    public void Cancel()
    {
        if (!IsActive)
            return;

        Target.Restore(Snapshot);
        IsActive = false;
        Target.Deselect();
    }
}

public abstract class EditorState
{
    public abstract void Enter(PropEditSession session);

Move State

오브젝트 이동 상태

public sealed class MovePropState : EditorState
{
    private readonly PlacementAreaResolver _areaResolver;
    private PropEditSession _session;
    private IMoveableProp _moveable;

    public override void Enter(PropEditSession session)
    {
        _session = session;

        if (!session.Target.TryGetMoveable(out _moveable))
            session.Cancel();
    }

    public override void Tick(EditorPointerInput input)
    {
        if (input.Cancelled)
        {
            _session.Cancel();
            return;
        }

        if (!_areaResolver.TryGetPlacementArea(
                input.Position, _session.Target, out var area))
            return;

        var nextPosition = CalculatePlacementPosition(area);
        if (_moveable.CanPlaceAt(nextPosition, area))
            _moveable.Move(nextPosition);

        if (input.Released)
            _session.Apply();
    }

    public override void Exit()
    {
        _session = null;
        _moveable = null;
    }
}

Rotate & UI

회전 상태와 편집 UI 이벤트 연결

public sealed class RotatePropState : EditorState
{
    private PropEditSession _session;
    private IRotatableProp _rotatable;
    private float _accumulatedAngle;

    public override void Enter(PropEditSession session)
    {
        _session = session;
        if (!session.Target.TryGetRotatable(out _rotatable))
            session.Cancel();
    }

    public override void Tick(EditorPointerInput input)
    {
        if (input.Cancelled)
        {
            _session.Cancel();
            return;
        }

        _accumulatedAngle += input.Delta.x * 0.35f;
        _rotatable.Rotate(Quaternion.Euler(0f, _accumulatedAngle, 0f));

        if (input.Released)
            _session.Apply();
    }

    public override void Exit()
    {
        _session = null;
        _rotatable = null;
        _accumulatedAngle = 0f;
    }
}
