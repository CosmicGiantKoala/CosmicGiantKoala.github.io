타겟팅과 데미지 계산의 입력 모델

public interface ITarget
{
    Vector3 Position { get; }
    bool IsAlive { get; }
}

public enum TargetingType
{
    None,
    MousePosition,
    Nearest,
    Random,
    Self
}

public readonly struct TargetingOption
{
    public TargetingType Type { get; }
    public int Count { get; }
    public float Range { get; }

    public TargetingOption(TargetingType type, int count, float range)
    {
        Type = type;
        Count = Math.Max(1, count);
        Range = Math.Max(0f, range);
    }
}

public readonly struct ActorStats
{
    public float AttackPower { get; init; }
    public float CriticalChance { get; init; }
    public float CriticalMultiplier { get; init; }
    public ElementBonus ElementBonus { get; init; }
}

public readonly struct SkillData
{
    public float PowerRate { get; init; }
    public ElementType Element { get; init; }
    public SkillSize Size { get; init; }
    public SkillAbility Ability { get; init; }
}

포인트 1. 인터페이스로 전투 대상 추상화

전투 시스템은 몬스터, 플레이어, 소환체 등 실제 구현체를 직접 알 필요가 없습니다. ITarget만 바라보게 하여 타겟 가능한 객체가 추가되어도 타겟팅 시스템 수정 범위를 줄입니다.

포인트 2. 계산 입력을 데이터로 고정

ActorStats와 SkillData를 계산 입력으로 분리하면 UI, 저장 데이터, 밸런스 테이블과 전투 계산 로직의 결합을 낮출 수 있습니다. 

public interface IRandomProvider
{
    float Value01();
    int Range(int minInclusive, int maxExclusive);
}

public sealed class UnityRandomProvider : IRandomProvider
{
    public float Value01() => UnityEngine.Random.value;
    public int Range(int minInclusive, int maxExclusive)
        => UnityEngine.Random.Range(minInclusive, maxExclusive);
}

랜덤 판정을 직접 호출하지 않고 인터페이스로 분리하면 치명타/랜덤 타겟 테스트를 안정적으로 작성할 수 있습니다.

Target System

타겟 등록, 탐색, 선택 흐름

public sealed class TargetSystem
{
    private readonly TargetScanner _scanner;
    private readonly HashSet<ITarget> _enemies = new();
    private ITarget _player;

    public TargetSystem(TargetScanner scanner)
    {
        _scanner = scanner;
    }

    public void InitPlayer(ITarget player)
    {
        _player = player ?? throw new ArgumentNullException(nameof(player));
    }

    public void AddEnemy(ITarget enemy)
    {
        if (enemy != null) _enemies.Add(enemy);
    }

    public void RemoveEnemy(ITarget enemy)
    {
        if (enemy != null) _enemies.Remove(enemy);
    }

    public IReadOnlyList<ITarget> FindTargets(
        Vector3 origin,
        TargetingOption option)
    {
        var aliveEnemies = _enemies.Where(enemy => enemy.IsAlive).ToArray();

        return option.Type switch
        {
            TargetingType.Self => _player != null ? new[] { _player } : Array.Empty<ITarget>(),
            TargetingType.Nearest => _scanner.FindNearest(origin, aliveEnemies, option.Count, option.Range),
            TargetingType.Random => _scanner.FindRandom(origin, aliveEnemies, option.Count, option.Range),
            _ => Array.Empty<ITarget>()
        };
    }
}

설계 의도

  • HashSet으로 중복 등록 방지
  • 사망한 타겟은 검색 대상에서 제외
  • 타겟 선택 알고리즘은 TargetScanner에 위임
  • 타겟 타입 추가 시 switch 분기만 확장 가능

실무 적용 지점

  • 마법, 스킬, 몬스터 AI가 같은 타겟 시스템을 재사용
  • 단일 타겟/다중 타겟을 같은 API로 처리
  • 플레이어 기준 탐색, 마우스 위치 탐색, 자기 자신 대상 처리 분리
  • 밸런스 데이터의 TargetingOption만 바꿔도 스킬 동작 변경 가능

Damage Calculator

데미지 정보 생성과 치명타 판정

public sealed class BattleDamageCalculator
{
    private const float Percent = 100f;
    private readonly IRandomProvider _random;

    public BattleDamageCalculator(IRandomProvider random)
    {
        _random = random;
    }

    public DamageInfo CreateDamageInfo(ActorStats actor, SkillData skill)
    {
        var baseDamage = CalculateBaseDamage(actor, skill);
        var criticalDamage = baseDamage * (actor.CriticalMultiplier / Percent);

        return new DamageInfo
        {
            CommonDamage = baseDamage,
            CriticalDamage = criticalDamage,
            CriticalChance = actor.CriticalChance,
            Element = skill.Element,
            Ability = CreateAbility(actor, skill)
        };
    }

    public DamageResult RollDamage(DamageInfo info)
    {
        if (info.CriticalChance <= 0f)
            return DamageResult.Normal(info.CommonDamage);

        var isCritical = _random.Value01() <= info.CriticalChance / Percent;
        return isCritical
            ? DamageResult.Critical(info.CriticalDamage)
            : DamageResult.Normal(info.CommonDamage);
    }

    private static float CalculateBaseDamage(ActorStats actor, SkillData skill)
    {
        var elementBonus = actor.ElementBonus.Get(skill.Element);
        var sizeBonus = actor.ElementBonus.Get(skill.Size);
        var multiplier = 1f + ((elementBonus + sizeBonus) / Percent);

        return actor.AttackPower * (skill.PowerRate * multiplier) / Percent;
    }
}

계산 책임 분리

전투 연출, 피격 처리, UI 갱신과 데미지 계산을 분리했습니다. 계산 클래스는 입력 데이터만 받아 결과를 반환하므로 테스트와 밸런스 검증이 쉽습니다.

확장 포인트

속성 보너스, 크기 보너스, 치명타, 부가 효과가 독립적으로 계산됩니다. 신규 속성이나 신규 부가 효과가 추가되어도 기존 계산 흐름을 유지하기 쉽습니다.

public sealed class FixedRandomProvider : IRandomProvider
{
    private readonly float _value;

    public FixedRandomProvider(float value)
    {
        _value = value;
    }

    public float Value01() => _value;
    public int Range(int minInclusive, int maxExclusive) => minInclusive;
}

// Example
// CriticalChance = 10%, random = 0.05 => critical
var calculator = new BattleDamageCalculator(new FixedRandomProvider(0.05f));
var info = calculator.CreateDamageInfo(actorStats, fireSkill);
var result = calculator.RollDamage(info);
