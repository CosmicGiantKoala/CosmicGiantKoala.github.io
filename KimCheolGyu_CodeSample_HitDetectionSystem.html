public readonly struct HitInfo
{
    public string AttackId { get; init; }
    public IHitReceiver Attacker { get; init; }
    public TeamId AttackerTeam { get; init; }
    public int Damage { get; init; }
    public float Stiffness { get; init; }
    public Vector3 Direction { get; init; }
    public float KnockbackPower { get; init; }
    public bool CanGuard { get; init; }
}

public readonly struct HitResult
{
    public bool Applied { get; init; }
    public bool Guarded { get; init; }
    public bool Killed { get; init; }
    public int AppliedDamage { get; init; }
    public float AppliedStiffness { get; init; }

    public static HitResult Ignored() => new() { Applied = false };
}

HitBox

공격 프레임 동안 히트박스 활성화

public sealed class HitBoxController : MonoBehaviour
{
    [SerializeField] private Collider hitCollider;
    private readonly HashSet<IHitReceiver> _hitTargets = new();
    private HitInfo _currentHitInfo;
    private bool _opened;

    public event Action<IHitReceiver, HitResult> HitApplied;

    public void Open(HitInfo hitInfo)
    {
        _currentHitInfo = hitInfo;
        _hitTargets.Clear();
        _opened = true;
        hitCollider.enabled = true;
    }

    public void Close()
    {
        _opened = false;
        hitCollider.enabled = false;
        _hitTargets.Clear();
    }

    private void OnTriggerEnter(Collider other)
    {
        if (!_opened) return;
        if (!other.TryGetComponent(out IHurtBox hurtBox)) return;
        TryApplyHit(hurtBox);
    }
}

실무 의도

애니메이션 이벤트나 스킬 타이밍에서 Open/Close를 호출해 공격 판정 프레임을 명확히 제어합니다.

Duplicate Guard

동일 공격 중복 피격 방지

private void TryApplyHit(IHurtBox hurtBox)
{
    var receiver = hurtBox.Receiver;

    if (receiver == null || !receiver.IsAlive)
        return;

    if (_currentHitInfo.AttackerTeam == receiver.Team)
        return;

    if (_hitTargets.Contains(receiver))
        return;

    _hitTargets.Add(receiver);

    var result = receiver.ReceiveHit(_currentHitInfo);
    if (result.Applied)
        HitApplied?.Invoke(receiver, result);
}

문제

한 번의 공격에서 여러 콜라이더가 같은 캐릭터에 닿으면 데미지가 중복 적용될 수 있습니다.

해결

공격 활성화 기간 동안 피격된 IHitReceiver를 HashSet으로 보관합니다.

Resolver

가드/무적/경직 판정

public sealed class HitResolver
{
    public HitResult Resolve(HitInfo hit, HeroStatus status)
    {
        if (status.IsDead || status.IsInvincible)
            return HitResult.Ignored();

        if (status.IsGuarding && hit.CanGuard)
        {
            status.ApplyGuardStiffness(hit.Stiffness * 0.35f);
            return new HitResult
            {
                Applied = true,
                Guarded = true,
                AppliedDamage = 0,
                AppliedStiffness = hit.Stiffness * 0.35f
            };
        }

        status.ApplyDamage(hit.Damage);
        status.ApplyStiffness(hit.Stiffness);
        status.ApplyKnockback(hit.Direction, hit.KnockbackPower);

        return new HitResult
        {
            Applied = true,
            Guarded = false,
            Killed = status.IsDead,
            AppliedDamage = hit.Damage,
            AppliedStiffness = hit.Stiffness
        };
    }
}

Status

피격 결과를 캐릭터 상태에 반영

public sealed class HeroStatus : MonoBehaviour, IHitReceiver
{
    [SerializeField] private int maxHp = 100;
    private int _hp;
    private float _stiffness;

    public event Action<int> HpChanged;
    public event Action<HitResult> Damaged;
    public event Action Died;

    public bool IsAlive => _hp > 0;
    public bool IsDead => _hp <= 0;
    public bool IsInvincible { get; private set; }
    public bool IsGuarding { get; private set; }

    public HitResult ReceiveHit(HitInfo hitInfo)
    {
        var result = _resolver.Resolve(hitInfo, this);
        if (!result.Applied) return result;

        Damaged?.Invoke(result);
        if (result.Killed) Died?.Invoke();
        return result;
    }

    public void ApplyDamage(int damage)
    {
        _hp = Mathf.Max(0, _hp - damage);
        HpChanged?.Invoke(_hp);
    }
}
